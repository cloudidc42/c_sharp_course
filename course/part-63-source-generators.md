# Part 63: Source Generators & Roslyn Analyzers — Compile-Time Code Generation (Steps 1639-1652)

## Steps 1639-1652: Mastering Roslyn Source Generators and Code Analyzers

Source Generators allow you to inspect user code and generate additional C# files at compile time — zero runtime reflection, zero overhead. Combined with Roslyn Analyzers and Code Fix Providers, you can enforce team conventions, auto-generate boilerplate, and build truly first-class developer tooling.

## Step 1640: Incremental Generators — The Modern Approach

`IIncrementalGenerator` is **strongly preferred** over `ISourceGenerator`. It caches intermediate results and only re-runs the parts of the pipeline that actually changed, dramatically improving IDE responsiveness.

```csharp
using System.Collections.Immutable;
using System.Text;
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp.Syntax;
using Microsoft.CodeAnalysis.Text;

namespace SourceGen;

[Generator]
public class AutoPropertyChangedGenerator : IIncrementalGenerator
{
    // Marker attribute source — inject it into the compilation
    private const string AttributeSource = """
        // <auto-generated/>
        namespace SourceGen.Attributes;
        
        [System.AttributeUsage(System.AttributeTargets.Class, Inherited = false)]
        public sealed class AutoNotifyAttribute : System.Attribute { }
        """;

    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        // Step 1: Register the attribute so user can apply it
        context.RegisterPostInitializationOutput(ctx =>
            ctx.AddSource("AutoNotifyAttribute.g.cs",
                SourceText.From(AttributeSource, Encoding.UTF8)));

        // Step 2: Find all classes marked with [AutoNotify]
        IncrementalValuesProvider<ClassDeclarationSyntax> classDeclarations =
            context.SyntaxProvider
                .ForAttributeWithMetadataName(
                    "SourceGen.Attributes.AutoNotifyAttribute",
                    predicate: static (node, _) => node is ClassDeclarationSyntax,
                    transform: static (ctx, _) => (ClassDeclarationSyntax)ctx.TargetNode)
                .Where(static cls => cls is not null)!;

        // Step 3: Combine with compilation for semantic model access
        IncrementalValueProvider<(Compilation, ImmutableArray<ClassDeclarationSyntax>)> compilationAndClasses =
            context.CompilationProvider.Combine(classDeclarations.Collect());

        // Step 4: Generate code
        context.RegisterSourceOutput(compilationAndClasses, static (spc, source) =>
            Execute(source.Item1, source.Item2, spc));
    }

    private static void Execute(
        Compilation compilation,
        ImmutableArray<ClassDeclarationSyntax> classes,
        SourceProductionContext context)
    {
        if (classes.IsDefaultOrEmpty) return;

        foreach (var classDecl in classes)
        {
            var model = compilation.GetSemanticModel(classDecl.SyntaxTree);
            if (model.GetDeclaredSymbol(classDecl) is not INamedTypeSymbol classSymbol) continue;

            var source = GenerateNotifyClass(classSymbol, classDecl);
            context.AddSource($"{classSymbol.Name}.AutoNotify.g.cs",
                SourceText.From(source, Encoding.UTF8));
        }
    }

    private static string GenerateNotifyClass(
        INamedTypeSymbol classSymbol,
        ClassDeclarationSyntax classDecl)
    {
        var namespaceName = classSymbol.ContainingNamespace.IsGlobalNamespace
            ? null
            : classSymbol.ContainingNamespace.ToDisplayString();

        // Find private fields to expose as notifying properties
        var fields = classSymbol.GetMembers()
            .OfType<IFieldSymbol>()
            .Where(f => f.DeclaredAccessibility == Accessibility.Private
                     && !f.IsStatic
                     && f.Name.StartsWith("_"))
            .ToList();

        var sb = new StringBuilder();
        sb.AppendLine("// <auto-generated/>");
        sb.AppendLine("#nullable enable");
        sb.AppendLine();
        sb.AppendLine("using System.ComponentModel;");
        sb.AppendLine();

        if (namespaceName is not null)
        {
            sb.AppendLine($"namespace {namespaceName};");
            sb.AppendLine();
        }

        sb.AppendLine($"partial class {classSymbol.Name} : INotifyPropertyChanged");
        sb.AppendLine("{");
        sb.AppendLine("    public event PropertyChangedEventHandler? PropertyChanged;");
        sb.AppendLine();
        sb.AppendLine("    private void OnPropertyChanged(string propertyName)");
        sb.AppendLine("        => PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(propertyName));");
        sb.AppendLine();

        foreach (var field in fields)
        {
            var fieldName = field.Name;
            // _firstName -> FirstName
            var propertyName = char.ToUpper(fieldName[1]) + fieldName.Substring(2);
            var typeName = field.Type.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat);

            sb.AppendLine($"    public {typeName} {propertyName}");
            sb.AppendLine("    {");
            sb.AppendLine($"        get => {fieldName};");
            sb.AppendLine("        set");
            sb.AppendLine("        {");
            sb.AppendLine($"            if ({fieldName} == value) return;");
            sb.AppendLine($"            {fieldName} = value;");
            sb.AppendLine($"            OnPropertyChanged(nameof({propertyName}));");
            sb.AppendLine("        }");
            sb.AppendLine("    }");
            sb.AppendLine();
        }

        sb.AppendLine("}");
        return sb.ToString();
    }
}
```

## Step 1641: `ForAttributeWithMetadataName` — The Right Way to Find Annotated Symbols

```csharp
[Generator]
public class MapperGenerator : IIncrementalGenerator
{
    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        // Register the [Mapper] attribute
        context.RegisterPostInitializationOutput(ctx => ctx.AddSource(
            "MapperAttribute.g.cs",
            SourceText.From("""
                // <auto-generated/>
                namespace SourceGen.Attributes;
                
                [System.AttributeUsage(System.AttributeTargets.Interface)]
                public sealed class MapperAttribute : System.Attribute { }
                """, Encoding.UTF8)));

        // ForAttributeWithMetadataName is the CORRECT, efficient way to find attributed symbols
        // It avoids the performance pitfall of filtering every syntax node
        var mapperInterfaces = context.SyntaxProvider
            .ForAttributeWithMetadataName(
                "SourceGen.Attributes.MapperAttribute",
                predicate: static (node, _) => node is InterfaceDeclarationSyntax,
                transform: static (ctx, ct) =>
                {
                    var symbol = (INamedTypeSymbol)ctx.TargetSymbol;
                    return new MapperInfo(
                        Symbol: symbol,
                        Methods: symbol.GetMembers()
                            .OfType<IMethodSymbol>()
                            .Select(m => new MapperMethod(
                                Name: m.Name,
                                ReturnType: m.ReturnType.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat),
                                ParameterType: m.Parameters.FirstOrDefault()?.Type
                                    .ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat) ?? "object"))
                            .ToImmutableArray());
                });

        context.RegisterSourceOutput(mapperInterfaces, static (spc, info) =>
        {
            var source = GenerateMapper(info);
            spc.AddSource($"{info.Symbol.Name}Impl.g.cs", SourceText.From(source, Encoding.UTF8));
        });
    }

    private static string GenerateMapper(MapperInfo info)
    {
        var ns = info.Symbol.ContainingNamespace.IsGlobalNamespace
            ? null : info.Symbol.ContainingNamespace.ToDisplayString();
        var sb = new StringBuilder();
        sb.AppendLine("// <auto-generated/>");
        sb.AppendLine("#nullable enable");
        sb.AppendLine();
        if (ns is not null) { sb.AppendLine($"namespace {ns};"); sb.AppendLine(); }

        sb.AppendLine($"internal sealed class {info.Symbol.Name}Impl : {info.Symbol.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat)}");
        sb.AppendLine("{");

        foreach (var method in info.Methods)
        {
            sb.AppendLine($"    public {method.ReturnType} {method.Name}({method.ParameterType} source)");
            sb.AppendLine("    {");
            sb.AppendLine("        ArgumentNullException.ThrowIfNull(source);");
            sb.AppendLine($"        var dest = new {method.ReturnType}();");
            sb.AppendLine("        // TODO: property mapping — extend generator to copy matching properties");
            sb.AppendLine("        return dest;");
            sb.AppendLine("    }");
            sb.AppendLine();
        }

        sb.AppendLine("}");
        return sb.ToString();
    }
}

record MapperInfo(INamedTypeSymbol Symbol, ImmutableArray<MapperMethod> Methods);
record MapperMethod(string Name, string ReturnType, string ParameterType);
```

## Step 1642: Advanced Generator — Strongly-Typed Route Constants

This generator scans all `[Route]` and `[HttpGet/Post/Put/Delete]` attributes in controllers and generates a `Routes` static class so you never mistype a URL.

```csharp
[Generator]
public class RouteConstantsGenerator : IIncrementalGenerator
{
    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        var routeData = context.SyntaxProvider
            .CreateSyntaxProvider(
                predicate: static (node, _) =>
                    node is ClassDeclarationSyntax cls &&
                    cls.AttributeLists.Count > 0 &&
                    cls.Identifier.Text.EndsWith("Controller"),
                transform: static (ctx, ct) => GetRouteInfo(ctx, ct))
            .Where(static info => info is not null)
            .Select(static (info, _) => info!);

        context.RegisterSourceOutput(routeData.Collect(), static (spc, routes) =>
        {
            if (routes.IsDefaultOrEmpty) return;

            var sb = new StringBuilder();
            sb.AppendLine("// <auto-generated/>");
            sb.AppendLine("public static class Routes");
            sb.AppendLine("{");

            foreach (var controller in routes)
            {
                var controllerName = controller.ControllerName;
                sb.AppendLine($"    public static class {controllerName}");
                sb.AppendLine("    {");
                sb.AppendLine($"        public const string Base = \"{controller.BaseRoute}\";");

                foreach (var action in controller.Actions)
                {
                    var fullRoute = string.IsNullOrEmpty(action.Route)
                        ? controller.BaseRoute
                        : $"{controller.BaseRoute}/{action.Route}".TrimEnd('/');
                    sb.AppendLine($"        public const string {action.Name} = \"{fullRoute}\";");
                }

                sb.AppendLine("    }");
            }

            sb.AppendLine("}");
            spc.AddSource("Routes.g.cs", SourceText.From(sb.ToString(), Encoding.UTF8));
        });
    }

    private static ControllerRouteInfo? GetRouteInfo(GeneratorSyntaxContext ctx, CancellationToken ct)
    {
        if (ctx.Node is not ClassDeclarationSyntax classDecl) return null;
        if (ctx.SemanticModel.GetDeclaredSymbol(classDecl, ct) is not INamedTypeSymbol symbol) return null;

        // Find [Route("...")] on the controller class
        var routeAttr = symbol.GetAttributes()
            .FirstOrDefault(a => a.AttributeClass?.Name is "RouteAttribute");
        if (routeAttr is null) return null;

        var baseRoute = routeAttr.ConstructorArguments.FirstOrDefault().Value?.ToString() ?? string.Empty;
        var controllerName = symbol.Name.Replace("Controller", "");

        var actions = symbol.GetMembers()
            .OfType<IMethodSymbol>()
            .Where(m => m.DeclaredAccessibility == Accessibility.Public && !m.IsStatic)
            .Select(m =>
            {
                var httpAttr = m.GetAttributes()
                    .FirstOrDefault(a => a.AttributeClass?.Name is
                        "HttpGetAttribute" or "HttpPostAttribute" or
                        "HttpPutAttribute" or "HttpDeleteAttribute" or "HttpPatchAttribute");
                var route = httpAttr?.ConstructorArguments.FirstOrDefault().Value?.ToString() ?? string.Empty;
                return new ActionRouteInfo(m.Name, route);
            })
            .ToImmutableArray();

        return new ControllerRouteInfo(controllerName, baseRoute, actions);
    }
}

record ControllerRouteInfo(string ControllerName, string BaseRoute, ImmutableArray<ActionRouteInfo> Actions);
record ActionRouteInfo(string Name, string Route);
```

## Step 1643: Generator for Discriminated Unions (DU Pattern)

```csharp
[Generator]
public class DiscriminatedUnionGenerator : IIncrementalGenerator
{
    private const string AttrSource = """
        // <auto-generated/>
        namespace SourceGen.Attributes;
        
        [System.AttributeUsage(System.AttributeTargets.Class)]
        public sealed class UnionAttribute : System.Attribute
        {
            public System.Type[] Cases { get; }
            public UnionAttribute(params System.Type[] cases) => Cases = cases;
        }
        """;

    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        context.RegisterPostInitializationOutput(ctx =>
            ctx.AddSource("UnionAttribute.g.cs", SourceText.From(AttrSource, Encoding.UTF8)));

        var unions = context.SyntaxProvider
            .ForAttributeWithMetadataName(
                "SourceGen.Attributes.UnionAttribute",
                predicate: static (n, _) => n is ClassDeclarationSyntax,
                transform: static (ctx, ct) =>
                {
                    var symbol = (INamedTypeSymbol)ctx.TargetSymbol;
                    var attr = symbol.GetAttributes()
                        .First(a => a.AttributeClass?.Name == "UnionAttribute");
                    var cases = attr.ConstructorArguments[0].Values
                        .Select(v => (INamedTypeSymbol)v.Value!)
                        .ToImmutableArray();
                    return (Symbol: symbol, Cases: cases);
                });

        context.RegisterSourceOutput(unions, static (spc, info) =>
        {
            var source = GenerateUnion(info.Symbol, info.Cases);
            spc.AddSource($"{info.Symbol.Name}.Union.g.cs", SourceText.From(source, Encoding.UTF8));
        });
    }

    private static string GenerateUnion(INamedTypeSymbol symbol, ImmutableArray<INamedTypeSymbol> cases)
    {
        var ns = symbol.ContainingNamespace.IsGlobalNamespace ? null
            : symbol.ContainingNamespace.ToDisplayString();
        var sb = new StringBuilder();
        sb.AppendLine("// <auto-generated/>");
        sb.AppendLine("#nullable enable");
        sb.AppendLine();
        if (ns is not null) { sb.AppendLine($"namespace {ns};"); sb.AppendLine(); }

        var typeName = symbol.Name;
        sb.AppendLine($"partial class {typeName}");
        sb.AppendLine("{");

        // Match method with one delegate per case
        var matchParams = string.Join(",\n        ",
            cases.Select(c => $"System.Func<{c.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat)}, TResult> on{c.Name}"));
        sb.AppendLine($"    public TResult Match<TResult>(\n        {matchParams})");
        sb.AppendLine("        => this switch");
        sb.AppendLine("        {");
        foreach (var c in cases)
        {
            var fqn = c.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat);
            sb.AppendLine($"            {fqn} x => on{c.Name}(x),");
        }
        sb.AppendLine("            _ => throw new System.InvalidOperationException($\"Unhandled union case: {this.GetType().Name}\")");
        sb.AppendLine("        };");
        sb.AppendLine();

        // Switch action (void match)
        var switchParams = string.Join(",\n        ",
            cases.Select(c => $"System.Action<{c.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat)}>? on{c.Name} = null"));
        sb.AppendLine($"    public void Switch(\n        {switchParams})");
        sb.AppendLine("    {");
        sb.AppendLine("        switch (this)");
        sb.AppendLine("        {");
        foreach (var c in cases)
        {
            var fqn = c.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat);
            sb.AppendLine($"            case {fqn} x: on{c.Name}?.Invoke(x); break;");
        }
        sb.AppendLine("        }");
        sb.AppendLine("    }");
        sb.AppendLine();

        // Is<T> helpers
        foreach (var c in cases)
        {
            var fqn = c.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat);
            sb.AppendLine($"    public bool Is{c.Name}([System.Diagnostics.CodeAnalysis.NotNullWhen(true)] out {fqn}? value)");
            sb.AppendLine("    {");
            sb.AppendLine($"        value = this as {fqn};");
            sb.AppendLine("        return value is not null;");
            sb.AppendLine("    }");
            sb.AppendLine();
        }

        sb.AppendLine("}");
        return sb.ToString();
    }
}
```

## Step 1644: Roslyn Analyzers — Enforce Team Conventions

Analyzers report diagnostics (errors/warnings/hints) in real-time in the IDE and in `dotnet build`.

```csharp
using System.Collections.Immutable;
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp;
using Microsoft.CodeAnalysis.CSharp.Syntax;
using Microsoft.CodeAnalysis.Diagnostics;

namespace SourceGen.Analyzers;

[DiagnosticAnalyzer(LanguageNames.CSharp)]
public class AsyncMethodNamingAnalyzer : DiagnosticAnalyzer
{
    private static readonly DiagnosticDescriptor Rule = new(
        id: "TEAM001",
        title: "Async method should end with 'Async'",
        messageFormat: "Method '{0}' is async but does not end with 'Async'",
        category: "Naming",
        defaultSeverity: DiagnosticSeverity.Warning,
        isEnabledByDefault: true,
        description: "Async methods must be named with the 'Async' suffix for clarity.");

    public override ImmutableArray<DiagnosticDescriptor> SupportedDiagnostics =>
        ImmutableArray.Create(Rule);

    public override void Initialize(AnalysisContext context)
    {
        // Required: disable concurrent execution and configure generated code analysis
        context.ConfigureGeneratedCodeAnalysis(GeneratedCodeAnalysisFlags.None);
        context.EnableConcurrentExecution();

        context.RegisterSyntaxNodeAction(AnalyzeMethodDeclaration, SyntaxKind.MethodDeclaration);
    }

    private static void AnalyzeMethodDeclaration(SyntaxNodeAnalysisContext context)
    {
        var method = (MethodDeclarationSyntax)context.Node;
        var symbol = context.SemanticModel.GetDeclaredSymbol(method, context.CancellationToken);

        if (symbol is null || !symbol.IsAsync) return;
        if (symbol.Name.EndsWith("Async")) return;

        // Suppress for interface implementations and overrides
        if (symbol.IsOverride || symbol.MethodKind == MethodKind.ExplicitInterfaceImplementation) return;

        var diagnostic = Diagnostic.Create(
            Rule,
            method.Identifier.GetLocation(),
            symbol.Name);

        context.ReportDiagnostic(diagnostic);
    }
}
```

```csharp
[DiagnosticAnalyzer(LanguageNames.CSharp)]
public class PrivateFieldNamingAnalyzer : DiagnosticAnalyzer
{
    private static readonly DiagnosticDescriptor Rule = new(
        id: "TEAM002",
        title: "Private field should start with underscore",
        messageFormat: "Private field '{0}' should be named '_{1}' (underscore prefix, camelCase)",
        category: "Naming",
        defaultSeverity: DiagnosticSeverity.Warning,
        isEnabledByDefault: true);

    public override ImmutableArray<DiagnosticDescriptor> SupportedDiagnostics =>
        ImmutableArray.Create(Rule);

    public override void Initialize(AnalysisContext context)
    {
        context.ConfigureGeneratedCodeAnalysis(GeneratedCodeAnalysisFlags.None);
        context.EnableConcurrentExecution();
        context.RegisterSymbolAction(AnalyzeField, SymbolKind.Field);
    }

    private static void AnalyzeField(SymbolAnalysisContext context)
    {
        var field = (IFieldSymbol)context.Symbol;

        if (field.DeclaredAccessibility != Accessibility.Private) return;
        if (field.IsConst || field.IsStatic) return;
        if (field.Name.StartsWith("_")) return;

        var suggestedName = "_" + char.ToLower(field.Name[0]) + field.Name.Substring(1);

        context.ReportDiagnostic(Diagnostic.Create(
            Rule,
            field.Locations[0],
            field.Name,
            suggestedName));
    }
}
```

```csharp
// TEAM003: Prohibit DateTime.Now — use DateTimeOffset.UtcNow instead
[DiagnosticAnalyzer(LanguageNames.CSharp)]
public class DateTimeNowAnalyzer : DiagnosticAnalyzer
{
    private static readonly DiagnosticDescriptor Rule = new(
        id: "TEAM003",
        title: "Do not use DateTime.Now",
        messageFormat: "Use 'DateTimeOffset.UtcNow' instead of 'DateTime.Now' to avoid timezone bugs",
        category: "Correctness",
        defaultSeverity: DiagnosticSeverity.Error,
        isEnabledByDefault: true);

    public override ImmutableArray<DiagnosticDescriptor> SupportedDiagnostics =>
        ImmutableArray.Create(Rule);

    public override void Initialize(AnalysisContext context)
    {
        context.ConfigureGeneratedCodeAnalysis(GeneratedCodeAnalysisFlags.None);
        context.EnableConcurrentExecution();
        context.RegisterSyntaxNodeAction(Analyze, SyntaxKind.SimpleMemberAccessExpression);
    }

    private static void Analyze(SyntaxNodeAnalysisContext context)
    {
        var memberAccess = (MemberAccessExpressionSyntax)context.Node;

        if (memberAccess.Name.Identifier.Text != "Now") return;

        var symbol = context.SemanticModel.GetSymbolInfo(memberAccess).Symbol;
        if (symbol?.ContainingType?.ToDisplayString() != "System.DateTime") return;

        context.ReportDiagnostic(Diagnostic.Create(Rule, memberAccess.GetLocation()));
    }
}
```

## Step 1645: Code Fix Providers

A Code Fix Provider pairs with an analyzer to offer an "Apply fix" action in the IDE.

```csharp
using System.Collections.Immutable;
using System.Composition;
using System.Threading;
using System.Threading.Tasks;
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CodeActions;
using Microsoft.CodeAnalysis.CodeFixes;
using Microsoft.CodeAnalysis.CSharp;
using Microsoft.CodeAnalysis.CSharp.Syntax;
using Microsoft.CodeAnalysis.Rename;

namespace SourceGen.CodeFixes;

[ExportCodeFixProvider(LanguageNames.CSharp, Name = nameof(AsyncMethodNamingFix))]
[Shared]
public class AsyncMethodNamingFix : CodeFixProvider
{
    public override ImmutableArray<string> FixableDiagnosticIds =>
        ImmutableArray.Create("TEAM001");

    public override FixAllProvider GetFixAllProvider() =>
        WellKnownFixAllProviders.BatchFixer;

    public override async Task RegisterCodeFixesAsync(CodeFixContext context)
    {
        var root = await context.Document
            .GetSyntaxRootAsync(context.CancellationToken)
            .ConfigureAwait(false);

        var diagnostic = context.Diagnostics[0];
        var diagnosticSpan = diagnostic.Location.SourceSpan;
        var declaration = root?.FindToken(diagnosticSpan.Start)
            .Parent?.AncestorsAndSelf()
            .OfType<MethodDeclarationSyntax>()
            .First();

        if (declaration is null) return;

        context.RegisterCodeFix(
            CodeAction.Create(
                title: "Rename method to add 'Async' suffix",
                createChangedSolution: ct => RenameAsync(context.Document, declaration, ct),
                equivalenceKey: "TEAM001_fix"),
            diagnostic);
    }

    private static async Task<Solution> RenameAsync(
        Document document,
        MethodDeclarationSyntax declaration,
        CancellationToken ct)
    {
        var semanticModel = await document.GetSemanticModelAsync(ct).ConfigureAwait(false);
        var methodSymbol = semanticModel?.GetDeclaredSymbol(declaration, ct);

        if (methodSymbol is null) return document.Project.Solution;

        var newName = methodSymbol.Name + "Async";
        var solution = document.Project.Solution;

        return await Renamer.RenameSymbolAsync(
            solution, methodSymbol, new SymbolRenameOptions(), newName, ct)
            .ConfigureAwait(false);
    }
}
```

```csharp
// Fix for TEAM003: Replace DateTime.Now with DateTimeOffset.UtcNow
[ExportCodeFixProvider(LanguageNames.CSharp, Name = nameof(DateTimeNowFix))]
[Shared]
public class DateTimeNowFix : CodeFixProvider
{
    public override ImmutableArray<string> FixableDiagnosticIds =>
        ImmutableArray.Create("TEAM003");

    public override FixAllProvider GetFixAllProvider() =>
        WellKnownFixAllProviders.BatchFixer;

    public override async Task RegisterCodeFixesAsync(CodeFixContext context)
    {
        var root = await context.Document
            .GetSyntaxRootAsync(context.CancellationToken)
            .ConfigureAwait(false);
        if (root is null) return;

        var diagnostic = context.Diagnostics[0];
        var node = root.FindNode(diagnostic.Location.SourceSpan);

        context.RegisterCodeFix(
            CodeAction.Create(
                title: "Replace with DateTimeOffset.UtcNow",
                createChangedDocument: ct => FixAsync(context.Document, node, ct),
                equivalenceKey: "TEAM003_fix"),
            diagnostic);
    }

    private static async Task<Document> FixAsync(
        Document document,
        SyntaxNode node,
        CancellationToken ct)
    {
        var root = await document.GetSyntaxRootAsync(ct).ConfigureAwait(false);
        if (root is null) return document;

        var replacement = SyntaxFactory.ParseExpression("DateTimeOffset.UtcNow")
            .WithTriviaFrom(node);

        var newRoot = root.ReplaceNode(node, replacement);
        return document.WithSyntaxRoot(newRoot);
    }
}
```

```xml
<!-- SourceGen.Tests.csproj -->
```

## Step 1648: Suppressor — Suppress Specific Analyzer Warnings

Suppressors allow you to silence warnings from other analyzers when your domain logic says they're safe.

```csharp
using System.Collections.Immutable;
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.Diagnostics;

namespace SourceGen.Suppressors;

// Suppress CA1822 ("Make member static") on methods decorated with [Benchmark]
[DiagnosticAnalyzer(LanguageNames.CSharp)]
public class BenchmarkMethodSuppressor : DiagnosticSuppressor
{
    private static readonly SuppressionDescriptor Rule = new(
        id: "SUPPRESS001",
        suppressedDiagnosticId: "CA1822",
        justification: "Benchmark methods must be instance methods for BenchmarkDotNet");

    public override ImmutableArray<SuppressionDescriptor> SupportedSuppressions =>
        ImmutableArray.Create(Rule);

    public override void ReportSuppressions(SuppressionAnalysisContext context)
    {
        foreach (var diagnostic in context.ReportedDiagnostics)
        {
            var location = diagnostic.Location;
            var node = location.SourceTree?.GetRoot(context.CancellationToken)
                .FindNode(location.SourceSpan);

            if (node is null) continue;

            var symbol = context.GetSemanticModel(location.SourceTree!)
                .GetDeclaredSymbol(node, context.CancellationToken);

            if (symbol is not IMethodSymbol method) continue;

            var hasBenchmarkAttr = method.GetAttributes()
                .Any(a => a.AttributeClass?.Name == "BenchmarkAttribute");

            if (hasBenchmarkAttr)
            {
                context.ReportSuppression(Suppression.Create(Rule, diagnostic));
            }
        }
    }
}
```

## Step 1650: Real-World Example — Auto-Generate EF Core DbContext Extensions

This generator scans entities marked with `[Entity]` and generates extension methods for `DbContext`:

```csharp
[Generator]
public class DbContextExtensionsGenerator : IIncrementalGenerator
{
    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        context.RegisterPostInitializationOutput(ctx => ctx.AddSource(
            "EntityAttribute.g.cs",
            SourceText.From("""
                // <auto-generated/>
                namespace SourceGen.Attributes;
                [System.AttributeUsage(System.AttributeTargets.Class)]
                public sealed class EntityAttribute : System.Attribute { }
                """, Encoding.UTF8)));

        var entities = context.SyntaxProvider
            .ForAttributeWithMetadataName(
                "SourceGen.Attributes.EntityAttribute",
                predicate: static (n, _) => n is ClassDeclarationSyntax,
                transform: static (ctx, _) =>
                {
                    var symbol = (INamedTypeSymbol)ctx.TargetSymbol;
                    var idProp = symbol.GetMembers()
                        .OfType<IPropertySymbol>()
                        .FirstOrDefault(p => p.Name == "Id");
                    return new EntityInfo(
                        symbol.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat),
                        symbol.Name,
                        symbol.ContainingNamespace.IsGlobalNamespace ? null
                            : symbol.ContainingNamespace.ToDisplayString(),
                        idProp?.Type.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat) ?? "global::System.Guid");
                });

        context.RegisterSourceOutput(entities.Collect(), GenerateExtensions);
    }

    private static void GenerateExtensions(
        SourceProductionContext spc,
        ImmutableArray<EntityInfo> entities)
    {
        if (entities.IsDefaultOrEmpty) return;

        var sb = new StringBuilder();
        sb.AppendLine("// <auto-generated/>");
        sb.AppendLine("#nullable enable");
        sb.AppendLine();
        sb.AppendLine("using System.Threading;");
        sb.AppendLine("using System.Threading.Tasks;");
        sb.AppendLine("using Microsoft.EntityFrameworkCore;");
        sb.AppendLine();
        sb.AppendLine("namespace Generated.Extensions;");
        sb.AppendLine();
        sb.AppendLine("public static class DbContextEntityExtensions");
        sb.AppendLine("{");

        foreach (var entity in entities)
        {
            sb.AppendLine($"    // Extensions for {entity.Name}");
            sb.AppendLine($"    public static async Task<{entity.FullyQualifiedName}?> Find{entity.Name}Async(");
            sb.AppendLine($"        this DbContext context, {entity.IdType} id, CancellationToken ct = default)");
            sb.AppendLine($"        => await context.Set<{entity.FullyQualifiedName}>().FindAsync([id], ct);");
            sb.AppendLine();

            sb.AppendLine($"    public static async Task<bool> {entity.Name}ExistsAsync(");
            sb.AppendLine($"        this DbContext context, {entity.IdType} id, CancellationToken ct = default)");
            sb.AppendLine($"        => await context.Set<{entity.FullyQualifiedName}>().AnyAsync(e => e.Id == id, ct);");
            sb.AppendLine();

            sb.AppendLine($"    public static async Task<int> Count{entity.Name}sAsync(");
            sb.AppendLine($"        this DbContext context, CancellationToken ct = default)");
            sb.AppendLine($"        => await context.Set<{entity.FullyQualifiedName}>().CountAsync(ct);");
            sb.AppendLine();
        }

        sb.AppendLine("}");
        spc.AddSource("DbContextEntityExtensions.g.cs", SourceText.From(sb.ToString(), Encoding.UTF8));
    }
}

record EntityInfo(string FullyQualifiedName, string Name, string? Namespace, string IdType);
```

## Step 1651: Performance Best Practices for Generators

```csharp
// WRONG: Scanning every syntax node is O(N) on every keypress in the IDE
var allClasses = context.SyntaxProvider
    .CreateSyntaxProvider(
        predicate: (node, _) => node is ClassDeclarationSyntax,   // fires on EVERY class
        transform: (ctx, _) => ctx.Node);

// RIGHT: Only visit nodes that have attributes — much smaller set
var annotatedClasses = context.SyntaxProvider
    .ForAttributeWithMetadataName(
        "MyNamespace.MyAttribute",
        predicate: static (node, _) => node is ClassDeclarationSyntax,
        transform: static (ctx, _) => (ClassDeclarationSyntax)ctx.TargetNode);
```

```csharp
// WRONG: Producing a large value that forces the whole pipeline to re-run when anything changes
var allSymbols = context.CompilationProvider
    .Select(static (compilation, _) => compilation.GetAllSymbols()); // huge, non-equatable

// RIGHT: Extract only what you need with a stable, equatable value
// Use .Select() to transform, and ensure your models implement IEquatable or are record types
// Records implement structural equality automatically
record MethodInfo(string ClassName, string MethodName, string ReturnType);

var methods = context.SyntaxProvider
    .ForAttributeWithMetadataName(...)
    .Select(static (ctx, _) => new MethodInfo(
        ctx.TargetSymbol.ContainingType.Name,
        ctx.TargetSymbol.Name,
        ((IMethodSymbol)ctx.TargetSymbol).ReturnType.Name));
// Records: structural equality means pipeline will be fully cached if nothing changed
```

```csharp
// WRONG: Using non-equatable types as pipeline values
// INamedTypeSymbol does NOT implement IEquatable — every run looks like a change
var symbols = context.SyntaxProvider
    .ForAttributeWithMetadataName(...)
    .Select(static (ctx, _) => (INamedTypeSymbol)ctx.TargetSymbol); // ← ISymbol, bad

// RIGHT: Extract primitive/equatable data from the symbol immediately
var data = context.SyntaxProvider
    .ForAttributeWithMetadataName(...)
    .Select(static (ctx, _) =>
    {
        var symbol = (INamedTypeSymbol)ctx.TargetSymbol;
        return new ClassData(
            symbol.Name,
            symbol.ContainingNamespace.ToDisplayString(),
            symbol.GetMembers().OfType<IPropertySymbol>()
                .Select(p => new PropData(p.Name, p.Type.ToDisplayString()))
                .ToImmutableArray()); // ImmutableArray<T> is equatable if T is equatable
    });
```

```csharp
// WRONG: Calling AddSource with duplicate file names crashes the generator host
context.AddSource("MyClass.g.cs", source1);
context.AddSource("MyClass.g.cs", source2); // ← ArgumentException

// RIGHT: Ensure unique hint names, typically by using the fully qualified type name
foreach (var classInfo in classes)
{
    var hintName = $"{classInfo.Namespace}.{classInfo.Name}.g.cs"
        .Replace('<', '_').Replace('>', '_').Replace(',', '_'); // sanitize generics
    spc.AddSource(hintName, source);
}
```

## Step 1652: Complete Real-World Generator — Auto-Logging Decorator

This generator creates a logging decorator for any interface annotated with `[AutoLog]`:

```csharp
[Generator]
public class AutoLogDecoratorGenerator : IIncrementalGenerator
{
    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        context.RegisterPostInitializationOutput(ctx => ctx.AddSource(
            "AutoLogAttribute.g.cs",
            SourceText.From("""
                // <auto-generated/>
                namespace SourceGen.Attributes;
                [System.AttributeUsage(System.AttributeTargets.Interface)]
                public sealed class AutoLogAttribute : System.Attribute { }
                """, Encoding.UTF8)));

        var interfaces = context.SyntaxProvider
            .ForAttributeWithMetadataName(
                "SourceGen.Attributes.AutoLogAttribute",
                predicate: static (n, _) => n is InterfaceDeclarationSyntax,
                transform: static (ctx, ct) => ExtractInterface(ctx, ct))
            .Where(static i => i is not null)
            .Select(static (i, _) => i!);

        context.RegisterSourceOutput(interfaces, static (spc, info) =>
        {
            var source = GenerateDecorator(info);
            spc.AddSource($"Logging{info.InterfaceName.TrimStart('I')}Decorator.g.cs",
                SourceText.From(source, Encoding.UTF8));
        });
    }

    private static InterfaceInfo? ExtractInterface(GeneratorAttributeSyntaxContext ctx, CancellationToken ct)
    {
        var symbol = (INamedTypeSymbol)ctx.TargetSymbol;
        var methods = symbol.GetMembers()
            .OfType<IMethodSymbol>()
            .Where(m => !m.IsStatic)
            .Select(m => new MethodData(
                m.Name,
                m.ReturnType.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat),
                m.ReturnType.Name is "Task" or "ValueTask" ||
                    (m.ReturnType is INamedTypeSymbol { IsGenericType: true } rt &&
                     rt.Name is "Task" or "ValueTask"),
                m.Parameters.Select(p => new ParameterData(
                    p.Type.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat),
                    p.Name))
                .ToImmutableArray()))
            .ToImmutableArray();

        return new InterfaceInfo(
            symbol.Name,
            symbol.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat),
            symbol.ContainingNamespace.IsGlobalNamespace ? null
                : symbol.ContainingNamespace.ToDisplayString(),
            methods);
    }

    private static string GenerateDecorator(InterfaceInfo info)
    {
        var sb = new StringBuilder();
        sb.AppendLine("// <auto-generated/>");
        sb.AppendLine("#nullable enable");
        sb.AppendLine();
        sb.AppendLine("using System;");
        sb.AppendLine("using System.Diagnostics;");
        sb.AppendLine("using System.Threading.Tasks;");
        sb.AppendLine("using Microsoft.Extensions.Logging;");
        sb.AppendLine();

        if (info.Namespace is not null) { sb.AppendLine($"namespace {info.Namespace};"); sb.AppendLine(); }

        var decoratorName = $"Logging{info.InterfaceName.TrimStart('I')}Decorator";
        sb.AppendLine($"internal sealed class {decoratorName} : {info.FullyQualifiedName}");
        sb.AppendLine("{");
        sb.AppendLine($"    private readonly {info.FullyQualifiedName} _inner;");
        sb.AppendLine($"    private readonly ILogger<{decoratorName}> _logger;");
        sb.AppendLine();
        sb.AppendLine($"    public {decoratorName}(");
        sb.AppendLine($"        {info.FullyQualifiedName} inner,");
        sb.AppendLine($"        ILogger<{decoratorName}> logger)");
        sb.AppendLine("    {");
        sb.AppendLine("        _inner = inner;");
        sb.AppendLine("        _logger = logger;");
        sb.AppendLine("    }");
        sb.AppendLine();

        foreach (var method in info.Methods)
        {
            var paramList = string.Join(", ", method.Parameters.Select(p => $"{p.TypeName} {p.Name}"));
            var argList = string.Join(", ", method.Parameters.Select(p => p.Name));
            var logArgs = method.Parameters.IsEmpty ? "" :
                ", " + string.Join(", ", method.Parameters.Select(p => p.Name));

            sb.AppendLine($"    public {(method.IsAsync ? "async " : "")}{method.ReturnType} {method.Name}({paramList})");
            sb.AppendLine("    {");
            sb.AppendLine($"        _logger.LogInformation(\"Calling {info.InterfaceName}.{method.Name}\"{logArgs});");
            sb.AppendLine("        var sw = Stopwatch.StartNew();");
            sb.AppendLine("        try");
            sb.AppendLine("        {");

            var callExpr = $"_inner.{method.Name}({argList})";
            if (method.ReturnType == "void")
            {
                sb.AppendLine($"            {callExpr};");
            }
            else if (method.IsAsync)
            {
                if (method.ReturnType.StartsWith("global::System.Threading.Tasks.Task<") ||
                    method.ReturnType.StartsWith("global::System.Threading.Tasks.ValueTask<"))
                {
                    sb.AppendLine($"            var result = await {callExpr};");
                    sb.AppendLine($"            _logger.LogInformation(\"{info.InterfaceName}.{method.Name} completed in {{Elapsed}}ms\", sw.ElapsedMilliseconds);");
                    sb.AppendLine("            return result;");
                }
                else
                {
                    sb.AppendLine($"            await {callExpr};");
                    sb.AppendLine($"            _logger.LogInformation(\"{info.InterfaceName}.{method.Name} completed in {{Elapsed}}ms\", sw.ElapsedMilliseconds);");
                }
            }
            else
            {
                sb.AppendLine($"            var result = {callExpr};");
                sb.AppendLine($"            _logger.LogInformation(\"{info.InterfaceName}.{method.Name} completed in {{Elapsed}}ms\", sw.ElapsedMilliseconds);");
                sb.AppendLine("            return result;");
            }

            sb.AppendLine("        }");
            sb.AppendLine("        catch (Exception ex)");
            sb.AppendLine("        {");
            sb.AppendLine($"            _logger.LogError(ex, \"Error in {info.InterfaceName}.{method.Name}\");");
            sb.AppendLine("            throw;");
            sb.AppendLine("        }");
            sb.AppendLine("    }");
            sb.AppendLine();
        }

        sb.AppendLine("}");

        // Also generate a DI extension
        sb.AppendLine();
        sb.AppendLine("public static class ServiceCollectionExtensionsFor" + decoratorName);
        sb.AppendLine("{");
        sb.AppendLine("    public static Microsoft.Extensions.DependencyInjection.IServiceCollection");
        sb.AppendLine($"        AddLogging{info.InterfaceName.TrimStart('I')}Decorator(");
        sb.AppendLine("        this Microsoft.Extensions.DependencyInjection.IServiceCollection services)");
        sb.AppendLine("    {");
        sb.AppendLine($"        services.Decorate<{info.FullyQualifiedName}, {decoratorName}>();");
        sb.AppendLine("        return services;");
        sb.AppendLine("    }");
        sb.AppendLine("}");

        return sb.ToString();
    }
}

record InterfaceInfo(
    string InterfaceName,
    string FullyQualifiedName,
    string? Namespace,
    ImmutableArray<MethodData> Methods);

record MethodData(
    string Name,
    string ReturnType,
    bool IsAsync,
    ImmutableArray<ParameterData> Parameters);

record ParameterData(string TypeName, string Name);
```

## Summary: Source Generators & Roslyn Analyzers — Key Patterns

| Pattern | When to use |
|---|---|
| `IIncrementalGenerator` | **Always** — prefer over `ISourceGenerator` |
| `ForAttributeWithMetadataName` | Finding annotated types/methods efficiently |
| `RegisterPostInitializationOutput` | Injecting marker attributes |
| Equatable record models | Ensure pipeline caching works correctly |
| `DiagnosticAnalyzer` | Enforce naming, patterns, prohibit APIs |
| `CodeFixProvider` | Pair with analyzers for one-click fixes |
| `DiagnosticSuppressor` | Silence false positives from other analyzers |
| Generator testing | Use `CSharpSourceGeneratorTest<TGen, DefaultVerifier>` |
| Analyzer testing | Use `CSharpAnalyzerVerifier<TAnalyzer>` |
| NuGet packaging | `IncludeBuildOutput=false`, place in `analyzers/dotnet/cs` |
